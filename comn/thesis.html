            return self.openpose_model

class ImageProcessor:
    def __init__(self):
        self.supported_formats = {'png', 'jpg', 'jpeg', 'gif', 'mp4', 'avi', 'mov'}
        
    def process_image(self, image_path):
        # 图像预处理
        image = cv2.imread(image_path)
        if image is None:
            raise ValueError("无法读取图像")
            
        # 图像增强
        enhanced = self.enhance_image(image)
        
        # 尺寸调整
        resized = self.resize_image(enhanced)
        
        return resized
        
    def process_video(self, video_path):
        # 视频处理
        cap = cv2.VideoCapture(video_path)
        frames = []
        
        while cap.isOpened():
            ret, frame = cap.read()
            if not ret:
                break
                
            # 处理每一帧
            processed_frame = self.process_frame(frame)
            frames.append(processed_frame)
            
        return frames

5.4 手语动作识别

系统使用YOLOv8和OpenPose进行手语动作识别，实现了完整的识别流程：

# 手语动作识别类
class SignLanguageRecognizer:
    def __init__(self, yolo_model, openpose_model):
        self.yolo_model = yolo_model
        self.openpose_model = openpose_model
        
    def recognize(self, image):
        # 1. 目标检测
        detections = self.yolo_model.predict(image)
        
        # 2. 关键点提取
        keypoints = self.extract_keypoints(image, detections)
        
        # 3. 动作识别
        actions = self.recognize_actions(keypoints)
        
        # 4. 结果后处理
        results = self.post_process(actions)
        
        return results
        
    def extract_keypoints(self, image, detections):
        # 使用OpenPose提取关键点
        blob = cv2.dnn.blobFromImage(image, 1.0/255, (368, 368))
        self.openpose_model.setInput(blob)
        output = self.openpose_model.forward()
        
        return self.process_keypoints(output)

# 结果输出处理类
class ResultProcessor:
    def __init__(self):
        self.tts_engine = self.init_tts()
        
    def process_result(self, recognition_result):
        # 1. 文字处理
        text_result = self.format_text(recognition_result)
        
        # 2. 语音合成
        if self.tts_enabled:
            self.speak_text(text_result)
            
        # 3. 可视化处理
        visual_result = self.visualize_result(recognition_result)
        
        return {
            'text': text_result,
            'visual': visual_result
        }
        
    def speak_text(self, text):
        try:
            self.tts_engine.say(text)
            self.tts_engine.runAndWait()
            return True
        except Exception as e:
            print(f"语音合成失败: {str(e)}")
            return False

5.5 用户界面设计

为了提高用户体验并简化操作过程，系统将图像输入、手语动作识别和结果输出集成到直观且用户友好的界面中。通过该用户界面，用户可以方便地选择输入源（摄像头、视频或图片），并进行模型选择、参数设置等操作。系统将输入传送给手语动作识别模块进行处理后，手语动作识别模块使用预先训练的模型识别输入中的手语动作，并对图像进行标注。用户可以通过界面直观地查看显示识别结果，如手语动作类型、置信度等信息，全面了解识别状态。该界面还支持语音播报识别结果，方便用户进行后续的交互和沟通。

整个用户界面分为左侧控制面板和右侧显示区域，如图5-5所示。左侧控制面板包含检测模式选择、模型控制、参数设置等功能区域；右侧显示区域包含视频显示区、操作按钮区、识别结果显示区和完整句子显示区。

图5-5 手语识别系统用户界面布局

5.5.1 检测模式选择区

检测模式选择区位于左侧控制面板的顶部，提供了三种输入源选择：摄像头检测、视频检测和图片检测。用户可以根据实际需求选择合适的输入源：

  • 摄像头检测：通过电脑摄像头实时捕获视频流，适用于实时手语识别场景。
  • 视频检测：支持上传本地视频文件进行手语识别，适用于离线分析场景。
  • 图片检测：支持上传单张图片进行手语识别，适用于静态图像分析场景。

5.5.2 模型控制区

模型控制区位于检测模式选择区的下方，提供了对识别模型的启用/禁用控制和模型选择功能：

  • OpenPose模型控制：通过复选框控制是否启用OpenPose模型进行姿态估计，并提供模型选择按钮。
  • YOLOv8模型控制：通过复选框控制是否启用YOLOv8模型进行手语动作识别，并提供模型选择按钮。

5.5.3 参数设置区

参数设置区位于模型控制区的下方，提供了多个参数调节滑块，用于优化识别效果：

  • 检测间隔：控制摄像头模式下进行手语识别的频率，范围从0.1秒到1.0秒。
  • 播放速度：控制视频播放的帧率，范围从1FPS到60FPS。
  • 置信度阈值：控制YOLOv8模型识别结果的置信度阈值，范围从0到100%。
  • IOU阈值：控制YOLOv8模型识别结果的IOU阈值，范围从0到100%。

5.5.4 视频显示区

视频显示区位于右侧显示区域的顶部，用于实时显示摄像头捕获的视频流、上传的视频文件或图片，以及识别结果的标注。该区域具有以下特点：

  • 支持实时显示摄像头捕获的视频流，帧率可达30FPS。
  • 支持显示上传的视频文件，播放速度可调。
  • 支持显示上传的图片及识别结果。
  • 在视频/图片上绘制识别结果，包括手语动作边界框、置信度和关键点标注。

5.5.5 操作按钮区

操作按钮区位于视频显示区的下方，提供了多个功能按钮，用于控制系统的运行：

  • 打开摄像头/关闭摄像头：控制摄像头的开启和关闭。
  • 选择视频/选择图片：根据当前选择的检测模式，打开文件选择对话框，选择要识别的视频或图片文件。
  • 开始识别：开始进行手语识别。
  • 停止识别：停止当前的手语识别过程。
  • 手语翻译：打开手语翻译窗口，显示数字人手语翻译。

5.5.6 识别结果显示区

识别结果显示区位于操作按钮区的下方，用于显示当前识别到的手语动作：

  • 显示当前识别到的手语动作类型和置信度。

  • 提供声音选择功能，可以选择不同的语音进行播报。

5.5.7 完整句子显示区

完整句子显示区位于识别结果显示区的下方，用于显示累积识别的手语动作组成的完整句子：

5.5.9 用户操作流程

用户可以通过以下步骤使用手语识别系统：

  1. 选择检测模式：根据实际需求，选择摄像头检测、视频检测或图片检测模式。
  2. 配置模型：选择是否启用OpenPose和YOLOv8模型，并根据需要选择模型文件。
  3. 调整参数：根据实际场景，调整检测间隔、播放速度、置信度阈值和IOU阈值等参数。
  4. 输入源操作：
    • 摄像头模式：点击"打开摄像头"按钮启动摄像头，然后点击"开始识别"按钮开始识别。
    • 视频模式：点击"选择视频"按钮选择视频文件，然后点击"开始识别"按钮开始识别。
    • 图片模式：点击"选择图片"按钮选择图片文件，系统会自动进行识别。
  5. 查看识别结果：在视频显示区查看标注后的图像，在识别结果显示区查看识别到的手语动作。
  6. 语音播报：点击"语音播报"按钮，系统会将识别结果转换为语音输出。
  7. 查看完整句子：在完整句子显示区查看累积识别的手语动作组成的完整句子。
  8. 手语翻译：点击"手语翻译"按钮，打开手语翻译窗口，显示数字人手语翻译。

通过这种直观且用户友好的界面设计，用户可以方便地进行手语识别操作，查看识别结果，并进行语音播报和手语翻译，大大提高了系统的易用性和实用性。
